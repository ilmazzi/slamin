<VirtualHost *:443>
    ServerName slamin.local
    DocumentRoot "C:/xampp/htdocs/slamin/public"

    # Configurazione SSL
    SSLEngine on
    SSLCertificateFile "C:/xampp/apache/conf/ssl.crt/server.crt"
    SSLCertificateKeyFile "C:/xampp/apache/conf/ssl.key/server.key"

    # Configurazione directory
    <Directory "C:/xampp/htdocs/slamin/public">
        AllowOverride All
        Require all granted
        Options Indexes FollowSymLinks MultiViews
    </Directory>

    # Proxy WebSocket per WSS (corretto)
    ProxyPass /ws ws://127.0.0.1:8080/
    ProxyPassReverse /ws ws://127.0.0.1:8080/

    # Headers di sicurezza
    Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"
    Header always set X-Content-Type-Options nosniff
    Header always set X-Frame-Options DENY
    Header always set X-XSS-Protection "1; mode=block"

    # Log
    ErrorLog "logs/slamin.local-error.log"
    CustomLog "logs/slamin.local-access.log" combined

    # Timeout per WebSocket
    ProxyTimeout 300
    ProxyBadHeader Ignore
</VirtualHost>

# Redirect HTTP to HTTPS
<VirtualHost *:80>
    ServerName slamin.local
    Redirect permanent / https://slamin.local/
</VirtualHost>
