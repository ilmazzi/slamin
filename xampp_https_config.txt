# Configurazione HTTPS per XAMPP
# Aggiungi questo al file C:\xampp\apache\conf\extra\httpd-ssl.conf

# Abilita SSL e moduli necessari
LoadModule ssl_module modules/mod_ssl.so
LoadModule proxy_module modules/mod_proxy.so
LoadModule proxy_wstunnel_module modules/mod_proxy_wstunnel.so
LoadModule proxy_http_module modules/mod_proxy_http.so

# Configurazione Virtual Host HTTPS per slamin.local
<VirtualHost *:443>
    ServerName slamin.local
    DocumentRoot "C:/xampp/htdocs/slamin/public"
    
    # Configurazione SSL
    SSLEngine on
    SSLCertificateFile "C:/xampp/apache/conf/ssl.crt/server.crt"
    SSLCertificateKeyFile "C:/xampp/apache/conf/ssl.key/server.key"
    
    # Configurazione directory
    <Directory "C:/xampp/htdocs/slamin/public">
        AllowOverride All
        Require all granted
    </Directory>
    
    # Proxy WebSocket per Reverb
    ProxyPreserveHost On
    ProxyRequests Off
    
    # Proxy per WebSocket Reverb
    RewriteEngine On
    RewriteCond %{HTTP:Upgrade} websocket [NC]
    RewriteCond %{HTTP:Connection} upgrade [NC]
    RewriteRule ^/app/(.*)$ ws://localhost:8080/app/$1 [P,L]
    
    # Proxy per HTTP normale (fallback)
    ProxyPass /app/ http://localhost:8080/app/
    ProxyPassReverse /app/ http://localhost:8080/app/
    
    # Proxy per WebSocket WSS -> WS
    LoadModule proxy_module modules/mod_proxy.so
    LoadModule proxy_wstunnel_module modules/mod_proxy_wstunnel.so
    LoadModule proxy_http_module modules/mod_proxy_http.so
    
    # Proxy per Reverb WebSocket
    ProxyPass /app/ ws://localhost:8080/app/
    ProxyPassReverse /app/ ws://localhost:8080/app/
    
    # Proxy per broadcasting auth
    ProxyPass /broadcasting/ http://localhost:8080/broadcasting/
    ProxyPassReverse /broadcasting/ http://localhost:8080/broadcasting/
    
    # Log
    ErrorLog "logs/slamin.local-error.log"
    CustomLog "logs/slamin.local-access.log" combined
</VirtualHost>

# Redirect HTTP to HTTPS
<VirtualHost *:80>
    ServerName slamin.local
    Redirect permanent / https://slamin.local/
</VirtualHost>

# ISTRUZIONI:
# 1. Apri C:\xampp\apache\conf\httpd.conf
# 2. Cerca la riga "#Include conf/extra/httpd-ssl.conf"
# 3. Rimuovi il # per abilitare SSL
# 4. Aggiungi la configurazione sopra al file httpd-ssl.conf
# 5. Riavvia Apache da XAMPP Control Panel
